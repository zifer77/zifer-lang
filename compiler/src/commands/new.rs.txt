use anyhow::{Context, Result};
use std::fs;
use std::path::Path;

const MAIN_TEMPLATE: &str = r#"// main.zf — generated by `zifer new`
server on 3000 {
    route GET "/" {
        return { message: "Hello from Zifer" }
    }
}
"#;

const GITIGNORE_TEMPLATE: &str = "zifer_modules/\ndist/\n*.zfc\n.env\n";

pub fn execute(name: &str) -> Result<()> {
    let root = Path::new(name);
    fs::create_dir_all(root).with_context(|| format!("Could not create directory `{name}`"))?;

    fs::write(root.join("main.zf"), MAIN_TEMPLATE)?;
    fs::write(root.join(".gitignore"), GITIGNORE_TEMPLATE)?;
    fs::write(
        root.join("zifer.toml"),
        format!("[project]\nname = \"{name}\"\nversion = \"0.1.0\"\nentry = \"main.zf\"\n"),
    )?;

    println!("Created new Zifer project in ./{name}");
    println!();
    println!("  cd {name}");
    println!("  zifer run main.zf");
    Ok(())
}
